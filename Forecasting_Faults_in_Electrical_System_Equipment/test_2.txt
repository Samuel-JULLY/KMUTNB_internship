# --- Data Processing Function (from your script, refactored) ---
def process_csv_data(df_raw):
    """
    Processes the raw DataFrame to perform DGA analysis, model training, and prediction.

    Args:
        df_raw (pd.DataFrame): The raw DataFrame loaded from the CSV.

    Returns:
        dict: A dictionary containing all processed data and model results
              needed for the Dash application.
    """
    df = df_raw.copy()

    # Step 1: Data Preparation and Cleaning
    # Drop specified columns if they exist
    df = df.drop(columns=["Unnamed: 20", "Unnamed: 21"], errors='ignore')

    # Drop rows where 'Year Test' is NaN, then convert to int
    df = df.dropna(subset=["Year Test"])
    if df.empty:
        raise ValueError("Le fichier CSV ne contient pas de données valides après le nettoyage de 'Year Test'.")
    df["Year Test"] = df["Year Test"].astype(int)

    # Filter out rows where all specified columns are 0.0
    colonnes_a_verifier = [
        'MFG', 'KV', 'MVA', 'Age', 'Year Test', 'O2', 'N2', 'CO2', 'CO',
        'H2', 'CH4', 'C2H2', 'C2H4', 'C2H6', 'C3H6', 'C3H8', 'TCG', 'TEMP', 'WATER'
    ]
    # Ensure all columns exist before checking
    valid_cols_to_check = [col for col in colonnes_a_verifier if col in df.columns]
    if valid_cols_to_check:
        df = df[~df[valid_cols_to_check].eq(0.0).all(axis=1)]
    
    if df.empty:
        raise ValueError("Le fichier CSV ne contient pas de données valides après le filtrage des lignes à zéros.")

    # Calculate 'year_num' for temporal features
    df["year_num"] = df["Year Test"] - df["Year Test"].min()

    # Calculate Ratios
    for ratio, num, denom in zip(["R1", "R2", "R3", "R4", "R5"], ["CH4", "C2H2", "C2H2", "C2H6", "C2H4"], ["H2", "C2H4", "CH4", "C2H2", "C2H6"]):
        if num in df.columns and denom in df.columns:
            df[ratio] = df[num] / df[denom].replace(0, np.nan)
        else:
            df[ratio] = np.nan # Assign NaN if columns are missing

    # Calculate Duval_Total
    duval_gases = ["CH4", "C2H2", "C2H4"]
    if all(g in df.columns for g in duval_gases):
        df["Duval_Total"] = df[duval_gases].sum(axis=1).replace(0, np.nan)
    else:
        df["Duval_Total"] = np.nan

    # Step 2: Classic Classification Methods
    # Define classification functions
    def classify_IRM(row):
        # Check if necessary columns exist before accessing
        if not all(col in row for col in ["R1", "R2", "R5"]): return "Uncertain"
        R1, R2, R5 = row["R1"], row["R2"], row["R5"]
        if pd.isna(R1) or pd.isna(R2) or pd.isna(R5): return "Uncertain"
        if R1 < 0.1 and R5 < 0.2: return "PD"
        elif R1 < 0.1 and R2 > 1 and R5 > 1: return "PD"
        elif 0.1 <= R1 <= 1 and 0.6 <= R2 <= 2.5 and R5 > 2: return "D1"
        elif R5 < 1: return "F1"
        elif R1 > 1 and R2 < 0.1 and 1 <= R5 <= 4: return "T2"
        elif R1 > 1 and R2 < 0.2 and R5 > 4: return "T3"
        else: return "Uncertain"

    def classify_DRM(row):
        if not all(col in row for col in ["R1", "R2", "R3", "R4", "H2", "CH4", "C2H2", "C2H4"]): return "Uncertain"
        if pd.isna(row["R1"]) or pd.isna(row["R2"]) or pd.isna(row["R3"]) or pd.isna(row["R4"]): return "Uncertain"
        if not (row["H2"] > 100 and (row["CH4"] > 120 or row["C2H2"] > 1 or row["C2H4"] > 50)): return "Uncertain"
        R1, R2, R3, R4 = row["R1"], row["R2"], row["R3"], row["R4"]
        if R1 > 1 and R2 < 0.75 and R3 < 0.3 and R4 > 0.4: return "F1"
        elif R1 < 0.1 and R3 < 0.3 and R4 > 0.4: return "PD"
        elif 0.1 <= R1 <= 1 and R2 > 0.75 and R3 < 0.3 and R4 < 0.4: return "D2"
        else: return "Uncertain"

    def classify_RRM(row):
        if not all(col in row for col in ["R1", "R2", "R5", "CH4", "H2", "C2H2", "C2H4", "C2H6"]): return "Uncertain"
        if pd.isna(row["R1"]) or pd.isna(row["R2"]) or pd.isna(row["R5"]): return "Uncertain"
        # Check if all relevant gas columns exist and are non-negative
        relevant_gases = ["CH4", "H2", "C2H2", "C2H4", "C2H6"]
        if not all(gas in row for gas in relevant_gases) or any(row[gas] <= 0 for gas in relevant_gases if gas in row):
            return "Uncertain"
        R1, R2, R5 = row["R1"], row["R2"], row["R5"]
        if 0.1 <= R1 <= 1 and R2 < 0.1 and R5 < 1: return "F1"
        elif R1 < 0.1 and R2 < 0.1 and R5 < 1: return "PD"
        elif 0.1 <= R1 <= 1 and 0.1 <= R2 <= 3 and R5 > 3: return "D2"
        elif 0.1 <= R1 <= 1 and R2 < 0.1 and 1 <= R5 <= 3: return "T1"
        elif R1 > 1 and R2 < 0.1 and 1 <= R5 <= 3: return "T2"
        elif R1 > 1 and R2 < 0.1 and R5 > 3: return "T3"
        else: return "Uncertain"

    def classify_DTM(row):
        if not all(col in row for col in ["CH4", "C2H2", "C2H4"]): return "Uncertain"
        ch4, c2h2, c2h4 = row["CH4"], row["C2H2"], row["C2H4"]
        total = ch4 + c2h2 + c2h4
        if total == 0: return "Uncertain"
        pct_ch4 = 100 * ch4 / total
        pct_c2h2 = 100 * c2h2 / total
        pct_c2h4 = 100 * c2h4 / total
        if pct_c2h2 > 23 and pct_c2h4 < 40: return "PD"
        elif pct_c2h4 > 40 and pct_c2h2 < 23: return "T2"
        elif 23 <= pct_c2h2 <= 50 and 20 <= pct_c2h4 <= 50: return "D1"
        elif pct_c2h4 < 20 and pct_c2h2 < 23: return "T1"
        else: return "Uncertain"

    def classify_DPM(row):
        gases_dpm = ["CH4", "C2H2", "C2H4", "H2", "C2H6"]
        if not all(col in row for col in gases_dpm): return "Uncertain"
        total = row[gases_dpm].sum()
        if total == 0: return "Uncertain"
        pct_ch4 = 100 * row["CH4"] / total
        pct_c2h2 = 100 * row["C2H2"] / total
        pct_c2h4 = 100 * row["C2H4"] / total
        pct_h2 = 100 * row["H2"] / total
        pct_c2h6 = 100 * row["C2H6"] / total
        if pct_c2h2 > 20 and pct_h2 > 15: return "PD"
        elif pct_c2h4 > 40 and pct_c2h6 < 10: return "T1"
        elif pct_c2h6 > 10 and pct_ch4 > 15: return "T3"
        else: return "Uncertain"

    method_columns = ["IRM_fault", "DRM_fault", "RRM_fault", "DTM_fault", "DPM_fault"]
    classification_funcs = [classify_IRM, classify_DRM, classify_RRM, classify_DTM, classify_DPM]

    for name, func in zip(method_columns, classification_funcs):
        df[name] = df.apply(func, axis=1)

    # Step 3: Calculate Duval Coordinates (Cx, Cy)
    gases_duval = ["CH4", "C2H2", "C2H4", "H2", "C2H6"]
    # Filter for gases actually present in columns
    available_gases_duval = [g for g in gases_duval if g in df.columns]

    if not available_gases_duval:
        raise ValueError("Missing essential gas columns for Duval analysis.")
    
    df["Total_DP"] = df[available_gases_duval].sum(axis=1)
    df = df[df["Total_DP"] > 0].copy() # Use .copy() to avoid SettingWithCopyWarning
    
    if df.empty:
        raise ValueError("Le fichier CSV ne contient pas de données valides après le filtrage pour 'Total_DP' > 0.")

    for g in available_gases_duval:
        df[f"%{g}"] = 100 * df[g] / df["Total_DP"]

    # Ensure necessary columns for Cx, Cy exist
    cx_components = ["%CH4", "%C2H6", "%C2H2", "%C2H4"]
    cy_components = ["%C2H6", "%C2H2"]
    if all(comp in df.columns for comp in cx_components) and all(comp in df.columns for comp in cy_components):
        df["Cx"] = (df["%CH4"] + 0.5 * df["%C2H6"] - 0.5 * df["%C2H2"] - df["%C2H4"]) / 100
        df["Cy"] = (0.866 * df["%C2H6"] + 0.866 * df["%C2H2"]) / 100
    else:
        # Assign NaNs or handle gracefully if components are missing
        df["Cx"] = np.nan
        df["Cy"] = np.nan


    # Step 4: Encoding and Target Preparation
    # Ensure method_columns exist in df before processing
    existing_method_columns = [col for col in method_columns if col in df.columns]
    if not existing_method_columns:
        raise ValueError("Aucune colonne de méthode de classification (IRM, DRM, etc.) n'a été générée. Vérifiez les données d'entrée.")

    # Calculate true_fault based on mode
    temp_df_for_mode = df[existing_method_columns].copy()
    
    # Replace any NaN values in classification columns with "Uncertain"
    # This prevents mode() from failing on NaNs and ensures "Uncertain" is considered.
    temp_df_for_mode = temp_df_for_mode.fillna("Uncertain")

    # If there are actual mode results, use them. Otherwise, default to "Uncertain".
    if not temp_df_for_mode.empty and not temp_df_for_mode.isnull().all().all():
        df["true_fault"] = temp_df_for_mode.mode(axis=1)[0]
    else:
        df["true_fault"] = "Uncertain" # Default if no data for mode calculation or all are NaN

    # Fit LabelEncoder *only* on the unique values actually present in the 'true_fault' column.
    # This ensures that fault_labels correctly maps existing data, and prevents empty classes.
    le_target = LabelEncoder()
    # Handle case where true_fault might be all 'Uncertain' or only one unique value
    if df["true_fault"].nunique() > 0:
        le_target.fit(df["true_fault"].unique())
        df["true_fault_index"] = le_target.transform(df["true_fault"])
    else:
        df["true_fault_index"] = np.nan # Or 0 if "Uncertain" is always implied as class 0
        le_target.classes_ = np.array([]) # Ensure it's not None or undefined

    # Create fault_labels based on the fitted LabelEncoder classes
    fault_labels = {i: c for i, c in enumerate(le_target.classes_)}
    print(f"DEBUG: fault_labels after LabelEncoder fit: {fault_labels}")
    print(f"DEBUG: Unique values in df['true_fault'] after mode: {df['true_fault'].unique()}")
    print(f"DEBUG: Unique values in df['true_fault_index'] after transform: {df['true_fault_index'].unique()}")


    # Temporal Feature Preparation
    # Check if 'Year Test' exists and is not empty
    if "Year Test" not in df.columns or df["Year Test"].empty:
        raise ValueError("La colonne 'Year Test' est manquante ou vide, impossible de créer des features temporelles.")
    
    df["year_sin"] = np.sin(2 * np.pi * df["Year Test"] / 12)
    df["year_cos"] = np.cos(2 * np.pi * df["Year Test"] / 12)

    X_temporal = df[["year_num", "year_sin", "year_cos"]]
    min_year_for_future = df["Year Test"].min() if "Year Test" in df.columns and not df["Year Test"].empty else 2000 # Default if no data

    # Check if y_temporal can be created
    if "true_fault_index" in df.columns and not df["true_fault_index"].isnull().all():
        y_temporal = df["true_fault_index"].dropna().values # Drop NaNs before converting to array
        # Ensure y_temporal is of integer type for model training
        if len(y_temporal) > 0:
            y_temporal = y_temporal.astype(int)
    else:
        y_temporal = np.array([]) # Empty array if no valid true_fault_index

    future_years = np.arange(min_year_for_future, 2031)
    future_year_num = future_years - min_year_for_future
    future_year_sin = np.sin(2 * np.pi * future_years / 12)
    future_year_cos = np.cos(2 * np.pi * future_years / 12)

    X_future_temporal = pd.DataFrame({
        "year_num": future_year_num,
        "year_sin": future_year_sin,
        "year_cos": future_year_cos
    })

    # Ensure 'true_fault_index' exists and is not empty for groupby
    real_proportions_by_year = pd.DataFrame() # Initialize as empty
    if "true_fault_index" in df.columns and not df.empty and df["true_fault_index"].nunique() > 0:
        # Cast to int to ensure correct grouping, dropna first
        df_temp_for_groupby = df.dropna(subset=["true_fault_index", "Year Test"]).copy()
        df_temp_for_groupby["true_fault_index"] = df_temp_for_groupby["true_fault_index"].astype(int)

        if not df_temp_for_groupby.empty:
            temp_real_proportions_raw = df_temp_for_groupby.groupby("Year Test")["true_fault_index"].value_counts(normalize=True).unstack(fill_value=0)
            
            # Ensure all possible fault labels from fault_labels are present as columns
            all_possible_fault_labels_names = [fault_labels[i] for i in sorted(fault_labels.keys())]
            
            index_for_full_df = temp_real_proportions_raw.index
            real_proportions_by_year = pd.DataFrame(0.0, index=index_for_full_df, columns=all_possible_fault_labels_names)
            
            for col_idx in temp_real_proportions_raw.columns:
                if col_idx in fault_labels: # Ensure the index is a valid fault_label key
                    col_name_str = fault_labels[col_idx]
                    real_proportions_by_year[col_name_str] = temp_real_proportions_raw[col_idx]
            print(f"DEBUG: real_proportions_by_year (after fillna and reindex): \n{real_proportions_by_year}")
        else:
            print("DEBUG: df_temp_for_groupby was empty, real_proportions_by_year remains empty.")
    else:
        # If no valid true_fault_index or empty df, create an empty dataframe with correct columns
        all_possible_fault_labels_names = [fault_labels[i] for i in sorted(fault_labels.keys())] if fault_labels else []
        real_proportions_by_year = pd.DataFrame(0.0, index=[], columns=all_possible_fault_labels_names)
        print("DEBUG: No valid true_fault_index or empty df, real_proportions_by_year initialized as empty.")


    # Model Definition and Hyperparameter Grids
    param_grids = {
        "Random Forest": {
            'n_estimators': [50, 100, 200],
            'max_depth': [5, 10, None],
            'min_samples_split': [2, 5]
        },
        "Naive Bayes": {},
        "KNN": {
            'n_neighbors': [3, 5, 7, 9],
            'weights': ['uniform', 'distance']
        },
        "SVM": {
            'C': [0.1, 1, 10],
            'kernel': ['rbf', 'linear'],
            'gamma': ['scale', 'auto']
        },
        "Logistic Regression": {
            'C': [0.1, 1, 10],
            'solver': ['liblinear', 'lbfgs'],
            'penalty': ['l2']
        },
        "Decision Tree": {
            'max_depth': [5, 10, None],
            'min_samples_split': [2, 5],
            'criterion': ['gini', 'entropy']
        }
    }

    models_to_evaluate = {
        "Random Forest": RandomForestClassifier(random_state=42),
        "Naive Bayes": GaussianNB(),
        "KNN": KNeighborsClassifier(),
        "SVM": SVC(probability=True, random_state=42),
        "Logistic Regression": LogisticRegression(max_iter=1000, random_state=42),
        "Decision Tree": DecisionTreeClassifier(random_state=42)
    }

    def evaluate_model_metrics(y_true, y_pred, average='macro'):
        # Handle cases with no true labels or only one class
        if len(y_true) == 0 or len(np.unique(y_true)) < 2:
            return 0, 0, 0, 0, 0 # Return zeros for all metrics

        acc = accuracy_score(y_true, y_pred)
        prec = precision_score(y_true, y_pred, average=average, zero_division=0)
        rec = recall_score(y_true, y_pred, average=average, zero_division=0)
        f1 = f1_score(y_true, y_pred, average=average, zero_division=0)

        cm = confusion_matrix(y_true, y_pred)
        specificity_per_class = []
        
        # Get unique classes present in y_true
        current_true_classes = np.unique(y_true)
        
        if not fault_labels or not hasattr(le_target, 'classes_') or len(le_target.classes_) == 0:
            return acc, prec, rec, 0, f1

        for current_class_int_idx in current_true_classes:
            if current_class_int_idx not in fault_labels:
                continue

            class_label_str = fault_labels[current_class_int_idx]

            try:
                cm_idx = np.where(le_target.classes_ == class_label_str)[0][0]
            except IndexError:
                specificity_per_class.append(0)
                continue

            if cm_idx < cm.shape[0] and cm_idx < cm.shape[1]:
                TP = cm[cm_idx, cm_idx]
                FN = cm[cm_idx, :].sum() - TP
                FP = cm[:, cm_idx].sum() - TP
                TN = cm.sum() - (TP + FN + FP)
                specificity = TN / (TN + FP) if (TN + FP) > 0 else 0
            else:
                specificity = 0
            specificity_per_class.append(specificity)
        
        specificity = np.mean(specificity_per_class) if specificity_per_class else 0

        return acc, prec, rec, specificity, f1

    # Temporal Prediction with ML Models
    scaler_temporal = StandardScaler()
    X_temporal_scaled_df = pd.DataFrame() 
    if not X_temporal.empty and not X_temporal.isnull().all().all():
        try:
            X_temporal_scaled = scaler_temporal.fit_transform(X_temporal)
            X_temporal_scaled_df = pd.DataFrame(X_temporal_scaled, columns=X_temporal.columns, index=X_temporal.index)
        except ValueError as e:
            print(f"Error scaling X_temporal: {e}. Proceeding without scaling.")
            X_temporal_scaled_df = X_temporal
    else:
        X_temporal_scaled_df = X_temporal

    unique_years_count = len(df["Year Test"].dropna().unique())
    tscv_splits = min(5, max(1, unique_years_count - 1))
    tscv = TimeSeriesSplit(n_splits=tscv_splits)

    results_temporal = {}
    all_model_predictions = {}
    best_temporal_models = {}

    for name, model_instance in models_to_evaluate.items():
        X_data_for_grid = X_temporal_scaled_df if name in ["SVM", "Logistic Regression", "KNN"] else X_temporal

        # Ensure enough samples and classes for training
        if len(X_data_for_grid) == 0 or len(np.unique(y_temporal)) < 2:
            results_temporal[name] = {
                "Accuracy": np.nan, "Precision": np.nan, "Recall": np.nan,
                "Specificity": np.nan, "F1": np.nan, "Best Params": "N/A (Insufficient Data)"
            }
            continue

        best_model = model_instance
        best_params = "N/A (paramètres par défaut)"

        if name in param_grids and param_grids[name]:
            try:
                grid_search = GridSearchCV(
                    estimator=model_instance,
                    param_grid=param_grids[name],
                    cv=tscv,
                    scoring='accuracy',
                    n_jobs=-1,
                    verbose=0
                )
                grid_search.fit(X_data_for_grid, y_temporal)
                best_model = grid_search.best_estimator_
                best_params = grid_search.best_params_
            except Exception as e:
                print(f"Error during GridSearchCV for {name} (temporal): {e}. Using default parameters.")
        
        best_temporal_models[name] = best_model

        accs, precs, recs, specs, f1s = [], [], [], [], []

        for fold_idx, (train_idx, test_idx) in enumerate(tscv.split(X_data_for_grid)):
            X_train_temp, X_test_temp = X_data_for_grid.iloc[train_idx], X_data_for_grid.iloc[test_idx]
            y_train_temp, y_test_temp = y_temporal[train_idx], y_temporal[test_idx]
            
            if len(np.unique(y_train_temp)) < 2 or len(np.unique(y_test_temp)) < 2:
                continue

            try:
                best_model.fit(X_train_temp, y_train_temp)
                y_pred_temp = best_model.predict(X_test_temp)
                
                acc, prec, rec, spec, f1 = evaluate_model_metrics(y_test_temp, y_pred_temp)
                accs.append(acc)
                precs.append(prec)
                recs.append(rec)
                specs.append(spec)
                f1s.append(f1)
            except Exception as e:
                print(f"Error during fold {fold_idx} evaluation for {name}: {e}")

        if accs:
            results_temporal[name] = {
                "Accuracy": np.mean(accs),
                "Precision": np.mean(precs),
                "Recall": np.mean(recs),
                "Specificity": np.mean(specs),
                "F1": np.mean(f1s),
                "Best Params": best_params
            }
        else:
            results_temporal[name] = {
                "Accuracy": np.nan, "Precision": np.nan, "Recall": np.nan,
                "Specificity": np.nan, "F1": np.nan, "Best Params": best_params
            }

        # Predict future proportions
        if len(np.unique(y_temporal)) >= 2 and fault_labels: # Ensure fault_labels is not empty
            try:
                best_model.fit(X_data_for_grid, y_temporal)
                if hasattr(best_model, "predict_proba"):
                    X_future_temporal_scaled = scaler_temporal.transform(X_future_temporal) if name in ["SVM", "Logistic Regression", "KNN"] else X_future_temporal
                    
                    proba_future = best_model.predict_proba(X_future_temporal_scaled)
                    
                    # Ensure all fault labels are represented in the prediction DataFrame
                    predicted_proportions_future_df = pd.DataFrame(0.0, index=future_years, columns=[fault_labels[i] for i in sorted(fault_labels.keys())])
                    
                    model_classes = best_model.classes_
                    # Map the model's internal class indices to our defined fault_labels
                    for i, class_idx_model in enumerate(model_classes):
                        # Find the string label for this model's internal class_idx_model
                        # and then use its position in our sorted global fault_labels to map proba
                        if class_idx_model in fault_labels:
                            label = fault_labels[class_idx_model]
                            if label in predicted_proportions_future_df.columns:
                                predicted_proportions_future_df[label] = proba_future[:, i]
                    
                    all_model_predictions[name] = predicted_proportions_future_df
                else:
                    print(f"Warning: Model {name} does not have predict_proba. Detailed proportion prediction skipped.")
            except Exception as e:
                print(f"Error during final training or future prediction for {name}: {e}. Future prediction skipped.")
        else:
            print(f"Warning: Not enough unique classes ({len(np.unique(y_temporal))}) or empty fault_labels for model {name} for future prediction.")

    metrics_df_temporal = pd.DataFrame(results_temporal).T.round(3)
    metrics_df_temporal.sort_values("Accuracy", ascending=False, inplace=True)

    best_temporal_model_name = ""
    if not metrics_df_temporal.empty and not metrics_df_temporal['Accuracy'].isnull().all():
        best_temporal_model_name = metrics_df_temporal['Accuracy'].idxmax()
    else:
        # Default to the first model name if no valid accuracy found or dataframe is empty
        best_temporal_model_name = list(models_to_evaluate.keys())[0] if models_to_evaluate else "No Model"

    return {
        "df": df,
        "real_proportions_by_year": real_proportions_by_year,
        "all_model_predictions": all_model_predictions,
        "metrics_df_temporal": metrics_df_temporal,
        "fault_labels": fault_labels,
        "best_temporal_model_name": best_temporal_model_name
    }


# --- Dash Application Setup ---
app = dash.Dash(__name__)
server = app.server

# Initial state for dropdowns and sliders before data is loaded

# --- Callback to process uploaded CSV and store results ---
@app.callback(
    Output('df-store', 'data'),
    Output('real-proportions-store', 'data'),
    Output('all-model-predictions-store', 'data'),
    Output('metrics-df-temporal-store', 'data'),
    Output('fault-labels-store', 'data'),
    Output('best-temporal-model-name-store', 'data'),
    Output('upload-message', 'children'),
    Output('available-kvs-store', 'data'),
    Output('available-mfgs-store', 'data'),
    Output('year-range-store', 'data'),
    Output('available-models-store', 'data'),
    Input('upload-data', 'contents'),
    State('upload-data', 'filename')
)
def upload_file(contents, filename):
    if contents is None:
        return (None, None, None, None, None, None, 
                "Veuillez charger un fichier CSV pour commencer l'analyse.",
                [], [], INITIAL_YEAR_RANGE, [])

    content_type, content_string = contents.split(',')
    decoded = base64.b64decode(content_string)

    try:
        if 'csv' in filename:
            # Assume that the user uploads a CSV file
            df_raw = pd.read_csv(io.StringIO(decoded.decode('utf-8')))
            
            # Process the data using the refactored script logic
            processed_data = process_csv_data(df_raw)

            # Extract results
            df = processed_data["df"]
            real_proportions_by_year = processed_data["real_proportions_by_year"]
            all_model_predictions = processed_data["all_model_predictions"]
            metrics_df_temporal = processed_data["metrics_df_temporal"]
            fault_labels = processed_data["fault_labels"]
            best_temporal_model_name = processed_data["best_temporal_model_name"]

            # Serialize DataFrames to JSON for storage
            df_json = df.to_json(orient='split', date_format='iso')
            real_proportions_json = real_proportions_by_year.to_json(orient='split', date_format='iso')
            metrics_df_temporal_json = metrics_df_temporal.to_json(orient='split', date_format='iso')
            
            # Serialize all_model_predictions (dict of DFs)
            all_model_predictions_json = {
                model_name: df_preds.to_json(orient='split', date_format='iso')
                for model_name, df_preds in all_model_predictions.items()
            }
            # Extract dropdown options based on the loaded data
            available_kvs = sorted(df["KV"].dropna().unique()) if "KV" in df.columns else []
            available_kvs_options = [{"label": "Tous", "value": "all"}] + [{"label": k, "value": k} for k in available_kvs]

            available_mfgs = sorted(df["MFG"].dropna().unique()) if "MFG" in df.columns else []
            available_mfgs_options = [{"label": "Tous", "value": "all"}] + [{"label": m, "value": m} for m in available_mfgs]

            # Get year range from data
            min_year = int(df["Year Test"].min()) if "Year Test" in df.columns and not df["Year Test"].empty else INITIAL_YEAR_RANGE[0]
            max_year = int(df["Year Test"].max()) if "Year Test" in df.columns and not df["Year Test"].empty else INITIAL_YEAR_RANGE[1]
            data_year_range = [min_year, max_year]
            
            available_models = list(all_model_predictions.keys())


            return (df_json, real_proportions_json, all_model_predictions_json, 
                    metrics_df_temporal_json, fault_labels, best_temporal_model_name,
                    f"Fichier '{filename}' traité avec succès.",
                    available_kvs_options, available_mfgs_options, data_year_range, available_models)
        
        else:
            return (None, None, None, None, None, None, 
                    "Erreur: Seuls les fichiers CSV sont acceptés.",
                    [], [], INITIAL_YEAR_RANGE, [])
    except Exception as e:
        print(f"Error processing file: {e}")
        return (None, None, None, None, None, None, 
                f"Erreur lors du traitement du fichier: {e}",
                [], [], INITIAL_YEAR_RANGE, [])

# --- Callback to update dropdowns and slider based on loaded data ---
@app.callback(
    Output('model-select', 'options'),
    Output('model-select', 'value'),
    Output('kv-select', 'options'),
    Output('kv-select', 'value'),
    Output('mfg-select', 'options'),
    Output('mfg-select', 'value'),
    Output('year-slider', 'min'),
    Output('year-slider', 'max'),
    Output('year-slider', 'value'),
    Output('year-slider', 'marks'),
    Output('year-slider', 'disabled'),
    Output('reset-button', 'n_clicks'), # Reset button clicks when data changes
    Input('df-store', 'data'), # Trigger when data is loaded
    Input('available-kvs-store', 'data'),
    Input('available-mfgs-store', 'data'),
    Input('year-range-store', 'data'),
    Input('available-models-store', 'data'),
    Input('best-temporal-model-name-store', 'data'),
)
def update_controls_on_data_load(df_json, available_kvs_options, available_mfgs_options, data_year_range, available_models, best_temporal_model_name):
    # Default values if no data is loaded yet
    if df_json is None:
        return ([], None, [], "all", [], "all", 
                INITIAL_YEAR_RANGE[0], INITIAL_YEAR_RANGE[1], INITIAL_YEAR_RANGE,
                {str(y): {'label': str(y), 'style': {'writing-mode': 'vertical-rl', 'text-orientation': 'sideways'}}
                    for y in range(INITIAL_YEAR_RANGE[0], INITIAL_YEAR_RANGE[1] + 1, 2)},
                True, 0) # Disable slider, reset clicks

    # Populate model options
    model_options = [{"label": m, "value": m} for m in available_models]
    model_value = best_temporal_model_name if best_temporal_model_name else (model_options[0]['value'] if model_options else None)

    # Populate KV and MFG options
    kv_options = available_kvs_options if available_kvs_options else [{"label": "Tous", "value": "all"}]
    mfg_options = available_mfgs_options if available_mfgs_options else [{"label": "Tous", "value": "all"}]

    # Update year slider range and marks
    min_year, max_year = data_year_range[0], data_year_range[1]
    slider_marks = {str(y): {'label': str(y), 'style': {'writing-mode': 'vertical-rl', 'text-orientation': 'sideways'}}
                    for y in range(min_year, max_year + 1, 2)}
    
    return (model_options, model_value, kv_options, "all", mfg_options, "all", 
            min_year, max_year, data_year_range, slider_marks, False, 0) # Enable slider, reset clicks

# --- Callback to update graph and metrics ---
@app.callback(
    Output('main-graph', 'figure'),
    Output('metrics-output', 'children'),
    Output('graph-title', 'children'),
    Input('model-select', 'value'),
    Input('kv-select', 'value'),
    Input('mfg-select', 'value'),
    Input('year-slider', 'value'),
    Input('reset-button', 'n_clicks'),
    State('df-store', 'data'),
    State('real-proportions-store', 'data'),
    State('all-model-predictions-store', 'data'),
    State('metrics-df-temporal-store', 'data'),
    State('fault-labels-store', 'data'),
    State('best-temporal-model-name-store', 'data'),
    State('year-range-store', 'data'), # To get original year range for reset
)
def update_graph(model_name, kv_selected, mfg_selected, year_range_slider, reset_clicks, 
                 df_json, real_proportions_json, all_model_predictions_json, 
                 metrics_df_temporal_json, fault_labels, best_temporal_model_name, initial_data_year_range):

    # If no data is loaded, display an empty graph and message
    if df_json is None:
        return (go.Figure(), 
                html.Div([html.H4("Aucune donnée chargée.", style={'color': '#555'}),
                          html.P("Veuillez charger un fichier CSV pour visualiser les données et les prédictions.")],
                         style={'textAlign': 'center', 'marginTop': '50px'}),
                "")

    # Determine if reset button was clicked
    ctx = dash.callback_context
    if not ctx.triggered:
        button_id = 'No clicks yet'
    else:
        button_id = ctx.triggered[0]['prop_id'].split('.')[0]

    # Deserialize data from dcc.Store
    df = pd.read_json(io.StringIO(df_json), orient='split')
    # Convert 'Year Test' to int after deserialization for consistent type handling
    if "Year Test" in df.columns:
        df["Year Test"] = df["Year Test"].astype(int)

    real_proportions_by_year = pd.read_json(io.StringIO(real_proportions_json), orient='split')
    metrics_df_temporal = pd.read_json(io.StringIO(metrics_df_temporal_json), orient='split')
    
    # Deserialize all_model_predictions
    all_model_predictions = {
        k: pd.read_json(io.StringIO(v), orient='split')
        for k, v in all_model_predictions_json.items()
    }

    # Handle reset button click
    if button_id == 'reset-button' and reset_clicks > 0:
        kv_selected = "all"
        mfg_selected = "all"
        # Reset year slider to the full range of the loaded data
        year_range_slider = initial_data_year_range 
        model_name = best_temporal_model_name if best_temporal_model_name else (list(all_model_predictions.keys())[0] if all_model_predictions else None)
    
    # Ensure model_name is not None after reset or initial load
    if model_name is None and all_model_predictions:
        model_name = list(all_model_predictions.keys())[0]

    # Filter df based on selected values
    filtered_df = df[df["Year Test"].between(*year_range_slider)].copy() # Use .copy() to avoid SettingWithCopyWarning
    
    if kv_selected != "all" and "KV" in filtered_df.columns:
        filtered_df = filtered_df[filtered_df["KV"] == kv_selected]
    
    if mfg_selected != "all" and "MFG" in filtered_df.columns:
        filtered_df = filtered_df[filtered_df["MFG"] == mfg_selected]

    traces = []

    # Real Proportions
    # Ensure filtered_df is not empty AND has the 'true_fault_index' column AND has at least one unique value in it
    if not filtered_df.empty and "true_fault_index" in filtered_df.columns and filtered_df["true_fault_index"].nunique() > 0:
        # Get actual counts of true_fault_index for the filtered DataFrame
        # Ensure 'true_fault_index' is integer type for groupby key consistency
        df_for_real_proportions = filtered_df.dropna(subset=["true_fault_index", "Year Test"]).copy()
        if not df_for_real_proportions.empty:
            df_for_real_proportions["true_fault_index"] = df_for_real_proportions["true_fault_index"].astype(int)
            temp_real_proportions_by_year = df_for_real_proportions.groupby("Year Test")["true_fault_index"].value_counts(normalize=True).unstack(fill_value=0)
            
            # Ensure all possible fault labels are present as columns
            all_fault_cols_sorted_indices = sorted(fault_labels.keys())
            all_fault_cols = [fault_labels[idx] for idx in all_fault_cols_sorted_indices]

            # Create a full DataFrame with all possible fault columns and fill missing with 0.0
            index_for_full_df = temp_real_proportions_by_year.index
            temp_real_proportions_by_year_full = pd.DataFrame(0.0, index=index_for_full_df, columns=all_fault_cols)
            
            # Copy data from temp_real_proportions_by_year to temp_real_proportions_by_year_full,
            # mapping original integer columns to their string labels
            for col_idx in temp_real_proportions_by_year.columns:
                if col_idx in fault_labels: # Only process if the index is a valid fault_label key
                    col_name_str = fault_labels[col_idx]
                    temp_real_proportions_by_year_full[col_name_str] = temp_real_proportions_by_year[col_idx]

            for fault in temp_real_proportions_by_year_full.columns:
                yearly = temp_real_proportions_by_year_full[fault]
                # Ensure there are actual values to plot (not all zeros or NaNs)
                if not yearly.empty and not yearly.isnull().all() and yearly.sum() > 0:
                    traces.append(go.Scatter(x=yearly.index, y=yearly.values,
                                             name=f"Réel {fault}", mode="lines+markers",
                                             line=dict(color='black', width=2),
                                             marker=dict(symbol='circle', size=7)))
        else:
            # filtered_df was not empty, but after dropping NaNs, df_for_real_proportions became empty
            min_year_display = df["Year Test"].min() if "Year Test" in df.columns and not df["Year Test"].empty else INITIAL_YEAR_RANGE[0]
            max_year_display = df["Year Test"].max() if "Year Test" in df.columns and not df["Year Test"].empty else INITIAL_YEAR_RANGE[1]
            traces.append(go.Scatter(x=[min_year_display, max_year_display], y=[0, 0],
                                     mode="lines", name="Aucune donnée réelle filtrée ou insuffisante pour le tracé (après NaN drop)"))

    else:
        # Fallback if no real data to plot, or insufficient unique values from the start
        min_year_display = df["Year Test"].min() if "Year Test" in df.columns and not df["Year Test"].empty else INITIAL_YEAR_RANGE[0]
        max_year_display = df["Year Test"].max() if "Year Test" in df.columns and not df["Year Test"].empty else INITIAL_YEAR_RANGE[1]

        traces.append(go.Scatter(x=[min_year_display, max_year_display], y=[0, 0],
                                 mode="lines", name="Aucune donnée réelle filtrée ou insuffisante pour le tracé"))


    # Model Predictions (always for all future years)
    if model_name and model_name in all_model_predictions:
        preds = all_model_predictions[model_name]
        
        last_train_year = df["Year Test"].max() if "Year Test" in df.columns and not df["Year Test"].empty else INITIAL_YEAR_RANGE[0]

        initial_prediction_points = {}
        # Only try to get initial prediction points if filtered_df is not empty
        if not filtered_df.empty and "Year Test" in filtered_df.columns and last_train_year in filtered_df["Year Test"].values:
            last_year_data = filtered_df[filtered_df["Year Test"] == last_train_year]
            if not last_year_data.empty and "true_fault_index" in last_year_data.columns:
                # Ensure the value_counts also normalize correctly and that fault_labels are aligned
                last_year_proportions_raw = (last_year_data["true_fault_index"].value_counts(normalize=True)).to_dict()
                for idx, prop in last_year_proportions_raw.items():
                    if idx in fault_labels: # Ensure the index is a valid fault label key
                        initial_prediction_points[fault_labels[idx]] = prop

        for fault in preds.columns:
            plot_preds = preds.copy()
            
            # Add an entry for last_train_year if it doesn't exist in predictions
            # and it's within or at the start of the prediction range
            if last_train_year not in plot_preds.index and last_train_year <= plot_preds.index.max() and last_train_year >= plot_preds.index.min():
                plot_preds.loc[last_train_year] = 0 # Temporary value, will be replaced
                plot_preds = plot_preds.sort_index()

            # Update the value for the last training year in predictions to match the real value
            if fault in initial_prediction_points and last_train_year in plot_preds.index:
                plot_preds.loc[last_train_year, fault] = initial_prediction_points.get(fault, 0)
            
            # Only add trace if there are actual prediction values (not all zeros or NaNs)
            if not plot_preds[fault].empty and not plot_preds[fault].isnull().all() and plot_preds[fault].sum() > 0:
                traces.append(go.Scatter(x=plot_preds.index, y=plot_preds[fault],
                                         name=f"{model_name} Prédit {fault}", mode="lines+markers",
                                         line=dict(dash='dash', width=2),
                                         marker=dict(symbol='x', size=7)))


    # Metrics Summary
    metrics_text = html.Div([
        html.H4("Résumé des performances du modèle sélectionné", style={'color': '#2C3E50', 'borderBottom': '1px solid #ccc', 'paddingBottom': '10px'}),
        html.P("Aucun modèle sélectionné ou données insuffisantes pour afficher les métriques.", style={'color': '#888'})
    ])

    if model_name and model_name in metrics_df_temporal.index:
        m = metrics_df_temporal.loc[model_name]
        metrics_text = html.Div([
            html.H4("Résumé des performances du modèle sélectionné", style={'color': '#2C3E50', 'borderBottom': '1px solid #ccc', 'paddingBottom': '10px'}),
            html.Ul([
                html.Li(f"Accuracy: {m.get('Accuracy', 'N/A'):.3f}", style={'marginBottom': '5px'}),
                html.Li(f"Precision: {m.get('Precision', 'N/A'):.3f}", style={'marginBottom': '5px'}),
                html.Li(f"Recall: {m.get('Recall', 'N/A'):.3f}", style={'marginBottom': '5px'}),
                html.Li(f"Specificity: {m.get('Specificity', 'N/A'):.3f}", style={'marginBottom': '5px'}),
                html.Li(f"F1-Score: {m.get('F1', 'N/A'):.3f}", style={'marginBottom': '5px'})
            ], style={'listStyleType': 'none', 'paddingLeft': '0'})
        ])
    else:
         metrics_text = html.Div([
            html.H4("Résumé des performances du modèle sélectionné", style={'color': '#2C3E50', 'borderBottom': '1px solid #ccc', 'paddingBottom': '10px'}),
            html.P("Aucun modèle sélectionné ou métriques non disponibles pour ce modèle.", style={'color': '#888'})
        ])


    title = f"Prédiction des proportions de défaillances par {model_name if model_name else 'Modèle non sélectionné'} (KV: {kv_selected}, MFG: {mfg_selected})"
    
    # Add vertical line to mark the end of training data
    shapes = []
    if "Year Test" in df.columns and not df["Year Test"].empty:
        last_train_year_plot = df["Year Test"].max()
        # Determine current x-axis range for plotting purposes
        # Safely get min/max from traces, fallback to initial_data_year_range if traces are empty
        x_values_in_traces = []
        for trace in traces:
            if hasattr(trace, 'x') and trace.x is not None:
                x_values_in_traces.extend(trace.x)

        current_x_min = min(x_values_in_traces) if x_values_in_traces else initial_data_year_range[0]
        current_x_max = max(x_values_in_traces) if x_values_in_traces else initial_data_year_range[1]

        if current_x_min <= last_train_year_plot <= current_x_max:
            shapes = [
                dict(
                    type="line",
                    x0=last_train_year_plot,
                    y0=0,
                    x1=last_train_year_plot,
                    y1=1,
                    xref="x",
                    yref="paper", # relative to the paper
                    line=dict(color="red", width=2, dash="dash"),
                    name="Fin des données d'entraînement"
                )
            ]

    # Determine x-axis range based on available data and predictions
    x_min_data = df["Year Test"].min() if "Year Test" in df.columns and not df["Year Test"].empty else INITIAL_YEAR_RANGE[0]
    x_max_pred = max(all_model_predictions[model_name].index) if model_name in all_model_predictions and not all_model_predictions[model_name].empty else INITIAL_YEAR_RANGE[1]

    # Take the minimum of data start year and max of prediction end year for overall x-axis range
    final_x_min = min(x_min_data, initial_data_year_range[0])
    final_x_max = max(x_max_pred, initial_data_year_range[1])


    figure = {
        'data': traces,
        'layout': go.Layout(
            title=title,
            xaxis=dict(title="Année", tickmode='linear', dtick=1, range=[final_x_min, final_x_max + 1]), # Adjusted range to show future years
            yaxis=dict(title="Proportion prédite", range=[0, 1.05]),
            hovermode='x unified',
            legend=dict(orientation="h", xanchor="center", x=0.5, y=-0.2),
            shapes=shapes
        )
    }

    return figure, metrics_text, title
